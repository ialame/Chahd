### Problème 1 : Combustion du butane d'un briquet
\begin{enonce}
Un briquet contient du butane $C_4H_{10}$ qui brûle dans le dioxygène de l'air selon :
$$2\,C_4H_{10} + 13\,O_2 \longrightarrow 8\,CO_2 + 10\,H_2O$$
On brûle $m = 2{,}9\ \text{g}$ de butane en présence de $n_0(O_2) = 0{,}40\ \text{mol}$ de dioxygène. On donne la masse molaire $M(C_4H_{10}) = 58\ \text{g}\cdot\text{mol}^{-1}$ et le volume molaire $V_m = 24\ \text{L}\cdot\text{mol}^{-1}$.
\begin{enumerate}
    \item Calculer la quantité de matière initiale de butane.
    \item Déterminer l'avancement maximal et le réactif limitant.
    \item Établir le bilan de matière à l'état final.
    \item Calculer le volume de dioxyde de carbone produit.
\end{enumerate}
\end{enonce}

\textbf{Solution :}
\begin{enumerate}
    \item $n_0(C_4H_{10}) = \dfrac{m}{M} = \dfrac{2{,}9}{58} = 0{,}050\ \text{mol}$.
    \item Le butane s'annule pour $x = \dfrac{n_0(C_4H_{10})}{2} = \dfrac{0{,}050}{2} = 0{,}025\ \text{mol}$ et le dioxygène pour $x = \dfrac{n_0(O_2)}{13} = \dfrac{0{,}40}{13} \approx 0{,}031\ \text{mol}$. Donc :
    $$x_{max} = \min(0{,}025\,;\ 0{,}031) = 0{,}025\ \text{mol}$$
    Le **butane** est le réactif limitant.
    \item Bilan de matière à l'état final :
    \begin{itemize}
        \item $n(C_4H_{10}) = 0{,}050 - 2 \times 0{,}025 = 0\ \text{mol}$ (limitant) ;
        \item $n(O_2) = 0{,}40 - 13 \times 0{,}025 = 0{,}40 - 0{,}325 = 0{,}075\ \text{mol}$ (excès) ;
        \item $n(CO_2) = 8 \times 0{,}025 = 0{,}20\ \text{mol}$ ;
        \item $n(H_2O) = 10 \times 0{,}025 = 0{,}25\ \text{mol}$.
    \end{itemize}
    \item $V(CO_2) = n(CO_2) \times V_m = 0{,}20 \times 24 = 4{,}8\ \text{L}$.
\end{enumerate}

### Problème 2 : Attaque du fer par l'acide chlorhydrique
\begin{enonce}
On plonge $m = 5{,}6\ \text{g}$ de fer dans une solution contenant $n_0(HCl) = 0{,}15\ \text{mol}$ de chlorure d'hydrogène. La transformation est modélisée par :
$$Fe + 2\,HCl \longrightarrow FeCl_2 + H_2$$
On donne $M(Fe) = 56\ \text{g}\cdot\text{mol}^{-1}$ et $V_m = 24\ \text{L}\cdot\text{mol}^{-1}$.
\begin{enumerate}
    \item Calculer la quantité de matière initiale de fer.
    \item Déterminer l'avancement maximal et le réactif limitant.
    \item Quelle masse de fer reste-t-il à l'état final ?
    \item Calculer le volume de dihydrogène dégagé.
\end{enumerate}
\end{enonce}

\textbf{Solution :}
\begin{enumerate}
    \item $n_0(Fe) = \dfrac{m}{M} = \dfrac{5{,}6}{56} = 0{,}10\ \text{mol}$.
    \item Le fer s'annule pour $x = \dfrac{n_0(Fe)}{1} = 0{,}10\ \text{mol}$ et l'acide pour $x = \dfrac{n_0(HCl)}{2} = \dfrac{0{,}15}{2} = 0{,}075\ \text{mol}$. Donc :
    $$x_{max} = \min(0{,}10\,;\ 0{,}075) = 0{,}075\ \text{mol}$$
    Le **chlorure d'hydrogène** ($HCl$) est le réactif limitant.
    \item À l'état final, $n(Fe) = 0{,}10 - 0{,}075 = 0{,}025\ \text{mol}$. La masse de fer restant vaut $m(Fe) = n(Fe) \times M = 0{,}025 \times 56 = 1{,}4\ \text{g}$.
    \item La quantité de dihydrogène formée est $n(H_2) = x_{max} = 0{,}075\ \text{mol}$, d'où $V(H_2) = 0{,}075 \times 24 = 1{,}8\ \text{L}$.
\end{enumerate}

### Problème 3 : Synthèse de l'eau, du mélange stœchiométrique à l'excès
\begin{enonce}
La synthèse de l'eau est modélisée par l'équation $2\,H_2 + O_2 \longrightarrow 2\,H_2O$. On donne $V_m = 24\ \text{L}\cdot\text{mol}^{-1}$.

\textbf{Partie A.} On dispose de $n_0(H_2) = 4{,}0\ \text{mol}$ de dihydrogène.
\begin{enumerate}
    \item Quelle quantité de dioxygène faut-il introduire pour réaliser un mélange stœchiométrique ?
    \item Déterminer $x_{max}$ et la quantité d'eau formée.
\end{enumerate}
\textbf{Partie B.} On reprend $n_0(H_2) = 4{,}0\ \text{mol}$ mais avec $n_0(O_2) = 3{,}0\ \text{mol}$.
\begin{enumerate}
    \item[3.] Déterminer le réactif limitant et l'avancement maximal.
    \item[4.] Établir le bilan de matière à l'état final.
    \item[5.] Calculer le volume de dioxygène consommé.
\end{enumerate}
\end{enonce}

\textbf{Solution :}
\textbf{Partie A.}
\begin{enumerate}
    \item Le mélange est stœchiométrique lorsque $\dfrac{n_0(H_2)}{2} = \dfrac{n_0(O_2)}{1}$, soit $n_0(O_2) = \dfrac{4{,}0}{2} = 2{,}0\ \text{mol}$.
    \item Les deux réactifs s'annulent ensemble pour $x_{max} = \dfrac{4{,}0}{2} = 2{,}0\ \text{mol}$. L'eau formée vaut $n(H_2O) = 2 \times 2{,}0 = 4{,}0\ \text{mol}$.
\end{enumerate}
\textbf{Partie B.}
\begin{enumerate}
    \item[3.] Le dihydrogène s'annule pour $x = \dfrac{4{,}0}{2} = 2{,}0\ \text{mol}$ et le dioxygène pour $x = \dfrac{3{,}0}{1} = 3{,}0\ \text{mol}$. Donc $x_{max} = \min(2{,}0\,;\ 3{,}0) = 2{,}0\ \text{mol}$ : le **dihydrogène** est le réactif limitant.
    \item[4.] Bilan : $n(H_2) = 4{,}0 - 2 \times 2{,}0 = 0\ \text{mol}$ ; $n(O_2) = 3{,}0 - 2{,}0 = 1{,}0\ \text{mol}$ (excès) ; $n(H_2O) = 2 \times 2{,}0 = 4{,}0\ \text{mol}$.
    \item[5.] La quantité de dioxygène consommée est $n_{cons}(O_2) = 1 \times x_{max} = 2{,}0\ \text{mol}$, d'où un volume $V = 2{,}0 \times 24 = 48\ \text{L}$.
\end{enumerate}
